// Bootstrap Nitemare 3D weapon layer.
//
// Selectors and ammo routing are recovered from the original runtime:
//   0 Single bolt plasma -> plasma ammo
//   1 Magic wand         -> wand ammo
//   2 Silver pistol      -> silver ammo (hitscan)
//   3 Multi bolt plasma  -> plasma ammo
//
// The current projectile velocity and damage presentation are intentionally
// bootstrap values. Exact Nitemare DDA/projected-damage behavior is a later
// native combat layer. ECWolf DECORATE accepts only half-tic fractions, so
// the recovered nominal 8 Hz attempt periods are represented by the nearest
// legal frame durations: 9 / 4.5 / 13 / 4.5.

actor NitemarePlayerProjectile
{
	radius 4
	speed 8
	damage 1
	PROJECTILE
	states
	{
		Spawn:
			TNT1 A 1
			loop
		Death:
			TNT1 A 1
			stop
	}
}

actor NitemarePlasmaShot : NitemarePlayerProjectile
{
}

actor NitemareWandShot : NitemarePlayerProjectile
{
}

actor NitemareWeapon : Weapon
{
	weapon.ammouse1 1
	weapon.ammogive1 0
	inventory.interhubamount 1
	+WEAPON.DONTBOB
	states
	{
		Select:
			TNT1 A 0 A_Raise
			loop
		Deselect:
			TNT1 A 0 A_Lower
			loop
	}
}

actor NitemareSinglePlasma : NitemareWeapon
{
	weapon.ammotype1 "NitemarePlasmaAmmo"
	weapon.slotnumber 1
	weapon.selectionorder 4000
	+WEAPON.NOAUTOFIRE
	states
	{
		Ready:
			TNT1 A 1 A_WeaponReady
			loop
		Fire:
			TNT1 A 0 A_NitemareFireMissile("NitemarePlasmaShot")
			TNT1 A 9
			goto Ready
	}
}

actor NitemareMultiPlasma : NitemareWeapon
{
	weapon.ammotype1 "NitemarePlasmaAmmo"
	weapon.slotnumber 4
	weapon.selectionorder 1000
	states
	{
		Ready:
			TNT1 A 1 A_WeaponReady
			loop
		Fire:
			TNT1 A 0 A_NitemareFireMissile("NitemarePlasmaShot")
			TNT1 A 4.5 A_ReFire
			goto Ready
		Hold:
			TNT1 A 0 A_NitemareFireMissile("NitemarePlasmaShot")
			TNT1 A 4.5 A_ReFire
			goto Ready
	}
}
